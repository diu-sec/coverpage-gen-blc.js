  const base = "https://cdn.jsdelivr.net/gh/diu-sec/coverpage-gen-blc.js/"

    const {
      patchZipInput
    } = await import(

    );

    const ls = JSON.parse(localStorage.coverpage);

    const data = {
      Semester: ls.course.semester,
      "Student Name": ls.student.name,
      "Student ID": ls.student.sid,
      Batch: "___________",
      Section: "__________",
      "Course Code": "SE 236",
      "Course Name": ls.course.name,
      "Course Teacher Name": ls.course.teacher,
      Designation: "Lecturer",
      "Submission Date": Date()
    };

    patchZipInput(`${base}/templates/coverpage.pdf`, data, true);
  }
